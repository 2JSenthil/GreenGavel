CREATE TABLE IF NOT EXISTS DOCUMENTS (
    document_id STRING,
    event_id STRING,
    meeting_name STRING,
    meeting_date STRING,
    meeting_time STRING,
    meeting_location STRING,
    document_type STRING,
    status STRING,
    source_url STRING,
    event_page_url STRING,
    full_text STRING
);

CREATE TABLE IF NOT EXISTS CHUNKS (
    chunk_id STRING,
    document_id STRING,
    event_id STRING,
    chunk_index INTEGER,
    chunk_text STRING,
    topics STRING,
    meeting_name STRING,
    meeting_date STRING,
    document_type STRING,
    status STRING,
    source_url STRING
);

USE DATABASE GREEN_GAVEL;

SELECT
    meeting_date,
    document_type,
    status,
    chunk_text,
    source_url
FROM CHUNKS
WHERE LOWER(chunk_text) LIKE '%stormwater%'
ORDER BY meeting_date DESC
LIMIT 10;

SELECT
    meeting_date,
    document_type,
    status,
    chunk_text,
    source_url
FROM CHUNKS
WHERE
    LOWER(chunk_text) LIKE '%stormwater%'
    OR LOWER(chunk_text) LIKE '%flood%'
    OR LOWER(chunk_text) LIKE '%drainage%'
    OR LOWER(chunk_text) LIKE '%watershed%'
ORDER BY meeting_date DESC
LIMIT 8;
